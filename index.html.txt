  <script>
    // --- 音效系統 (Web Audio API) ---
    class SoundSystem {
      constructor() {
        this.ctx = null;
      }

      init() {
        if (!this.ctx) {
          this.ctx = new (window.AudioContext || window.webkitAudioContext)();
        }
      }

      playJump() {
        if (!this.ctx) return;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.type = 'sine';
        osc.frequency.setValueAtTime(300, this.ctx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(600, this.ctx.currentTime + 0.15);
        gain.gain.setValueAtTime(0.3, this.ctx.currentTime);
        gain.gain.linearRampToValueAtTime(0, this.ctx.currentTime + 0.15);
        osc.start();
        osc.stop(this.ctx.currentTime + 0.15);
      }

      playCollect() {
        if (!this.ctx) return;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(523.25, this.ctx.currentTime); // C5
        osc.frequency.setValueAtTime(659.25, this.ctx.currentTime + 0.08); // E5
        gain.gain.setValueAtTime(0.2, this.ctx.currentTime);
        gain.gain.linearRampToValueAtTime(0, this.ctx.currentTime + 0.2);
        osc.start();
        osc.stop(this.ctx.currentTime + 0.2);
      }

      playHit() {
        if (!this.ctx) return;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(150, this.ctx.currentTime);
        osc.frequency.linearRampToValueAtTime(60, this.ctx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.4, this.ctx.currentTime);
        gain.gain.linearRampToValueAtTime(0, this.ctx.currentTime + 0.2);
        osc.start();
        osc.stop(this.ctx.currentTime + 0.2);
      }
    }

    const sound = new SoundSystem();

    // --- 明信片圖鑑資料庫 ---
    const POSTCARDS = [
      { id: 'forest', name: '櫻花森林', icon: '🌸', reqScore: 0, desc: '靜謐的日式粉色森林。' },
      { id: 'pyramid', name: '埃及金字塔', icon: '🏜️', reqScore: 300, desc: '雄偉的沙漠金字塔古蹟。' },
      { id: 'paris', name: '巴黎鐵塔', icon: '🗼', reqScore: 700, desc: '浪漫璀璨的夜間鐵塔。' },
      { id: 'space', name: '璀璨星空', icon: '🌌', reqScore: 1200, desc: '飛越天際的夢幻星空。' }
    ];

    // --- 遊戲邏輯與 Canvas 渲染 ---
    const canvas = document.getElementById('gameCanvas');
    const ctx = canvas.getContext('2d');

    let gameRunning = false;
    let score = 0;
    let foodCount = 0;
    let lives = 3;
    let gameSpeed = 4;
    let frameCount = 0;
    let unlockedPostcards = JSON.parse(localStorage.getItem('foxie_postcards') || '["forest"]');

    // 角色 (小狐狸)
    const fox = {
      x: 80,
      y: 300,
      width: 44,
      height: 38,
      velocityY: 0,
      gravity: 0.6,
      jumpForce: -11,
      isGrounded: false,
      jumpCount: 0,
      maxJumps: 2,
      legAngle: 0,

      reset() {
        this.y = 300;
        this.velocityY = 0;
        this.jumpCount = 0;
      },

      jump() {
        if (this.jumpCount < this.maxJumps) {
          this.velocityY = this.jumpForce;
          this.jumpCount++;
          sound.playJump();
        }
      },

      update() {
        this.velocityY += this.gravity;
        this.y += this.velocityY;

        const groundLevel = 350 - this.height;
        if (this.y >= groundLevel) {
          this.y = groundLevel;
          this.velocityY = 0;
          this.isGrounded = true;
          this.jumpCount = 0;
        } else {
          this.isGrounded = false;
        }

        this.legAngle += 0.2;
      },

      draw() {
        ctx.save();
        ctx.translate(this.x, this.y);

        // 狐狸身體 (橘色)
        ctx.fillStyle = '#ff7e5f';
        ctx.beginPath();
        ctx.ellipse(22, 22, 18, 14, 0, 0, Math.PI * 2);
        ctx.fill();

        // 蓬鬆尾巴 (帶白色尖端)
        ctx.fillStyle = '#ff7e5f';
        ctx.beginPath();
        ctx.ellipse(-2, 16, 12, 7, -0.3, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = '#ffffff';
        ctx.beginPath();
        ctx.ellipse(-10, 14, 5, 4, -0.3, 0, Math.PI * 2);
        ctx.fill();

        // 頭部
        ctx.fillStyle = '#ff7e5f';
        ctx.beginPath();
        ctx.arc(32, 12, 13, 0, Math.PI * 2);
        ctx.fill();

        // 白色臉頰
        ctx.fillStyle = '#ffffff';
        ctx.beginPath();
        ctx.arc(34, 16, 8, 0, Math.PI * 2);
        ctx.fill();

        // 狐狸耳朵
        ctx.fillStyle = '#ff7e5f';
        ctx.beginPath();
        ctx.moveTo(26, 2);
        ctx.lineTo(31, -8);
        ctx.lineTo(35, 2);
        ctx.fill();

        // 黑色眼睛與鼻子
        ctx.fillStyle = '#2c3e50';
        ctx.beginPath();
        ctx.arc(36, 10, 2, 0, Math.PI * 2); // 眼
        ctx.arc(43, 14, 2.5, 0, Math.PI * 2); // 鼻
        ctx.fill();

        // 腳步跑步動畫
        ctx.strokeStyle = '#e056fd';
        ctx.lineWidth = 3;
        const legOffset = Math.sin(this.legAngle) * 6;
        if (this.isGrounded) {
          ctx.beginPath();
          ctx.moveTo(14, 32); ctx.lineTo(14 + legOffset, 38);
          ctx.moveTo(30, 32); ctx.lineTo(30 - legOffset, 38);
          ctx.stroke();
        } else {
          // 跳躍姿勢
          ctx.beginPath();
          ctx.moveTo(14, 32); ctx.lineTo(10, 36);
          ctx.moveTo(30, 32); ctx.lineTo(36, 36);
          ctx.stroke();
        }

        ctx.restore();
      }
    };

    // 障礙物與收集品
    let obstacles = [];
    let items = [];

    function spawnEntities() {
      if (frameCount % 100 === 0) {
        // 生成障礙物
        obstacles.push({
          x: 820,
          y: 320,
          width: 24,
          height: 30,
          type: Math.random() > 0.5 ? 'wood' : 'stone'
        });
      }

      if (frameCount % 70 === 0) {
        // 生成美食收集品
        items.push({
          x: 820,
          y: 200 + Math.random() * 80,
          width: 22,
          height: 22,
          icon: Math.random() > 0.5 ? '🍎' : '🍕',
          collected: false
        });
      }
    }

    // --- 場景背景主題設定 ---
    function getCurrentScene() {
      if (score < 300) {
        return {
          name: '📍 櫻花森林',
          sky: '#fce4ec',
          ground: '#81c784',
          bgObj: '🌸'
        };
      } else if (score < 700) {
        return {
          name: '📍 埃及金字塔',
          sky: '#ffe0b2',
          ground: '#e0a96d',
          bgObj: '🏜️'
        };
      } else if (score < 1200) {
        return {
          name: '📍 巴黎鐵塔',
          sky: '#c5cae9',
          ground: '#7986cb',
          bgObj: '🗼'
        };
      } else {
        return {
          name: '📍 璀璨星空',
          sky: '#263238',
          ground: '#455a64',
          bgObj: '✨'
        };
      }
    }

    function checkPostcardUnlock() {
      POSTCARDS.forEach(card => {
        if (score >= card.reqScore && !unlockedPostcards.includes(card.id)) {
          unlockedPostcards.push(card.id);
          localStorage.setItem('foxie_postcards', JSON.stringify(unlockedPostcards));
        }
      });
    }

    // 繪製背景
    function drawBackground(scene) {
      // 天空
      ctx.fillStyle = scene.sky;
      ctx.fillRect(0, 0, canvas.width, canvas.height);

      // 背景裝飾物件 (遠景)
      ctx.font = '32px serif';
      ctx.globalAlpha = 0.3;
      const bgX = (800 - (frameCount * 0.5) % 900);
      ctx.fillText(scene.bgObj, bgX, 220);
      ctx.fillText(scene.bgObj, (bgX + 450) % 900, 180);
      ctx.globalAlpha = 1.0;

      // 地面
      ctx.fillStyle = scene.ground;
      ctx.fillRect(0, 350, canvas.width, 100);
      ctx.fillStyle = 'rgba(0,0,0,0.08)';
      ctx.fillRect(0, 350, canvas.width, 8);
    }

    // --- 遊戲主循環 ---
    function gameLoop() {
      if (!gameRunning) return;

      frameCount++;
      score += 0.2; // 隨著時間累積里程
      gameSpeed = 4 + Math.floor(score / 200) * 0.5;

      const scene = getCurrentScene();
      document.getElementById('location-display').innerText = scene.name;
      document.getElementById('score-display').innerText = `里程: ${Math.floor(score)} km | 🍎 ${foodCount}`;

      checkPostcardUnlock();

      ctx.clearRect(0, 0, canvas.width, canvas.height);
      drawBackground(scene);

      // 更新與繪製狐狸
      fox.update();
      fox.draw();

      spawnEntities();

      // 處理障礙物
      for (let i = obstacles.length - 1; i >= 0; i--) {
        let obs = obstacles[i];
        obs.x -= gameSpeed;

        // 繪製障礙物
        ctx.fillStyle = obs.type === 'wood' ? '#8d6e63' : '#78909c';
        ctx.beginPath();
        ctx.roundRect(obs.x, obs.y, obs.width, obs.height, 6);
        ctx.fill();

        // 碰撞偵測
        if (
          fox.x < obs.x + obs.width &&
          fox.x + fox.width > obs.x &&
          fox.y < obs.y + obs.height &&
          fox.y + fox.height > obs.y
        ) {
          sound.playHit();
          lives--;
          updateLivesDisplay();
          obstacles.splice(i, 1);

          if (lives <= 0) {
            endGame();
            return;
          }
          continue;
        }

        if (obs.x < -30) obstacles.splice(i, 1);
      }

      // 處理美食收集品
      for (let i = items.length - 1; i >= 0; i--) {
        let item = items[i];
        item.x -= gameSpeed;

        ctx.font = '20px serif';
        ctx.fillText(item.icon, item.x, item.y + 18);

        // 碰撞偵測
        if (
          fox.x < item.x + item.width &&
          fox.x + fox.width > item.x &&
          fox.y < item.y + item.height &&
          fox.y + fox.height > item.y
        ) {
          sound.playCollect();
          foodCount++;
          items.splice(i, 1);
          continue;
        }

        if (item.x < -30) items.splice(i, 1);
      }

      requestAnimationFrame(gameLoop);
    }

    function updateLivesDisplay() {
      let hearts = '';
      for (let i = 0; i < lives; i++) hearts += '❤️ ';
      document.getElementById('lives-display').innerText = hearts || '💔';
    }

    function startGame() {
      sound.init();
      gameRunning = true;
      score = 0;
      foodCount = 0;
      lives = 3;
      obstacles = [];
      items = [];
      frameCount = 0;
      fox.reset();
      updateLivesDisplay();

      document.getElementById('start-screen').classList.add('hidden');
      document.getElementById('gameover-screen').classList.add('hidden');
      document.getElementById('gallery-modal').classList.add('hidden');

      requestAnimationFrame(gameLoop);
    }

    function endGame() {
      gameRunning = false;
      document.getElementById('gameover-stats').innerHTML = `總里程：${Math.floor(score)} km<br>收集美食：${foodCount} 個`;
      document.getElementById('gameover-screen').classList.remove('hidden');
    }

    // --- 圖鑑介面邏輯 ---
    function openGallery() {
      const grid = document.getElementById('gallery-grid');
      grid.innerHTML = '';

      POSTCARDS.forEach(card => {
        const isUnlocked = unlockedPostcards.includes(card.id);
        const item = document.createElement('div');
        item.className = `gallery-item ${isUnlocked ? '' : 'locked'}`;
        item.innerHTML = `
          <div class="icon">${isUnlocked ? card.icon : '🔒'}</div>
          <div class="title">${isUnlocked ? card.name : '未解鎖'}</div>
          <div style="font-size:10px; color:#888; margin-top:2px;">${isUnlocked ? card.desc : `需要 ${card.reqScore} km`}</div>
        `;
        grid.appendChild(item);
      });

      document.getElementById('gallery-modal').classList.remove('hidden');
    }

    function closeGallery() {
      document.getElementById('gallery-modal').classList.add('hidden');
    }

    // --- 操作事件監聽 ---
    window.addEventListener('keydown', (e) => {
      if (e.code === 'Space' || e.code === 'ArrowUp') {
        if (gameRunning) {
          fox.jump();
        } else if (document.getElementById('start-screen').classList.contains('hidden') === false) {
          startGame();
        }
      }
    });

    canvas.addEventListener('touchstart', (e) => {
      e.preventDefault();
      if (gameRunning) fox.jump();
    });

    canvas.addEventListener('mousedown', () => {
      if (gameRunning) fox.jump();
    });
  </script>
